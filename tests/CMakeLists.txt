function(isim_test name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE isim_core Qt6::Test)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ISIM_DATA_DIR=${CMAKE_SOURCE_DIR}/data" TIMEOUT 600)
endfunction()
isim_test(test_core)
isim_test(test_simulation)
isim_test(test_ai)
